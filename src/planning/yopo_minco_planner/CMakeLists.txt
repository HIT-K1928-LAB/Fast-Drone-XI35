cmake_minimum_required(VERSION 3.10)
project(yopo_minco_planner)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
add_compile_options(-Wall -Wextra)
find_package(catkin REQUIRED COMPONENTS roscpp std_msgs sensor_msgs nav_msgs geometry_msgs mavros_msgs quadrotor_msgs)
find_package(Eigen3 REQUIRED)
find_package(OpenCV REQUIRED COMPONENTS core imgproc photo)
find_package(CUDA REQUIRED)
find_path(TENSORRT_INCLUDE_DIR NvInfer.h PATH_SUFFIXES aarch64-linux-gnu x86_64-linux-gnu)
find_library(TENSORRT_LIBRARY nvinfer PATH_SUFFIXES aarch64-linux-gnu x86_64-linux-gnu)
if(NOT TENSORRT_INCLUDE_DIR OR NOT TENSORRT_LIBRARY)
  message(FATAL_ERROR "TensorRT headers/library missing")
endif()
catkin_package(INCLUDE_DIRS include LIBRARIES yopo_minco_core CATKIN_DEPENDS roscpp std_msgs sensor_msgs nav_msgs geometry_msgs mavros_msgs quadrotor_msgs)
include_directories(include ${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS} ${OpenCV_INCLUDE_DIRS} ${CUDA_INCLUDE_DIRS} ${TENSORRT_INCLUDE_DIR})
add_library(yopo_minco_core src/minco_solver.cpp src/yopo_planner.cpp src/yopo_engine.cpp)
target_link_libraries(yopo_minco_core ${TENSORRT_LIBRARY} ${CUDA_CUDART_LIBRARY})
add_executable(yopo_minco_node src/yopo_node.cpp)
add_dependencies(yopo_minco_node ${catkin_EXPORTED_TARGETS})
target_link_libraries(yopo_minco_node yopo_minco_core ${catkin_LIBRARIES} ${OpenCV_LIBRARIES})
add_executable(yopo_minco_numeric_probe tests/numeric_probe.cpp)
target_link_libraries(yopo_minco_numeric_probe yopo_minco_core)
add_executable(yopo_minco_engine_probe tests/engine_probe.cpp)
target_link_libraries(yopo_minco_engine_probe yopo_minco_core)
install(TARGETS yopo_minco_core yopo_minco_node yopo_minco_numeric_probe yopo_minco_engine_probe
  ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY include/${PROJECT_NAME}/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
